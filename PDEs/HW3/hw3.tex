\item Find the solution of \eqref{WE} for $x \in \mathbb{R}, t > 0$ with 
$$u(0,x) = f(x) \quad \mathrm{and} \quad \frac{\partial u}{\partial t}(0,x) = 0$$
for $x \in \mathbb{R}$. In particular, write the solution in terms of the $f(x)$ function. Then, graph by hand $u(\frac{1}{c},x)$, $u(\frac{2}{c},x)$, and $u(\frac{3}{c},x)$ using different axes. \\ \\
\textcolor{blue}{\textbf{Solution} — To find the general solution of (WE), apply the substitution
\begin{align*}
& r(t, x) = x - ct \\
& s(t, x) = x + ct
\end{align*}
Assume $u(t,x) = U(r,s)$ is a solution and substitute (we first calculate the needed derivatives of $u$):
\begin{align*}
\frac{\partial u}{\partial x} &= \frac{\partial U}{\partial r} \frac{\partial r}{\partial x} + \frac{\partial U}{\partial s} \frac{\partial s}{\partial x} \\ \\
&= \frac{\partial U}{\partial r} +  \frac{\partial U}{\partial s}
\end{align*}
And 
\begin{align*}
\frac{\partial^2 u}{\partial x^2} &= \frac{\partial}{\partial x} \left( \frac{\partial U}{\partial r} +  \frac{\partial U}{\partial s}  \right) \\
\end{align*}
To simplify further, note that $\frac{\partial U}{\partial r}$ is some function of $r$ and $s$, say $\phi_1(r,s)$, and $\frac{\partial U}{\partial s}$ is also some function of $r$ and $s$, say $\phi_2(r,s)$:
\begin{align*}
&= \frac{\partial}{\partial x} \left( \phi_1(r,s) +  \phi_2(r,s)  \right) \\ \\
&= \frac{\partial \phi_1}{\partial r} \frac{\partial r}{\partial x} + \frac{\partial \phi_1}{\partial s} \frac{\partial s}{\partial x} + \frac{\partial \phi_2}{\partial r} \frac{\partial r}{\partial x} + \frac{\partial \phi_2}{\partial s} \\ \\
&= \frac{\partial \phi_1}{\partial r} + \frac{\partial \phi_1}{\partial s} + \frac{\partial \phi_2}{\partial r} + \frac{\partial \phi_2}{\partial s} \\ \\
&= \frac{\partial}{\partial r} \left( \frac{\partial U}{\partial r} \right) + \frac{\partial}{\partial s} \left( \frac{\partial U}{\partial r} \right) + \frac{\partial}{\partial r} \left( \frac{\partial U}{\partial s} \right) + \frac{\partial}{\partial s} \left( \frac{\partial U}{\partial s} \right) \\ \\
&= \frac{\partial^2 U}{\partial r^2} + \frac{\partial^2 U}{\partial s \partial r} + \frac{\partial^2 U}{\partial r \partial s} + \frac{\partial^2 U}{\partial s^2} 
\end{align*}
Let us assume that $U(r,s)$ is smooth and well-behaved enough for the cross-derivatives to be equal. Then
\begin{align*}
\frac{\partial^2 u}{\partial x^2} &= \frac{\partial^2 U}{\partial r^2} + 2 \left( \frac{\partial^2 U}{\partial r \partial s} \right) + \frac{\partial^2 U}{\partial s^2} 
\end{align*}
By similar mechanics, we obtain 
\begin{align*}
\frac{\partial^2 u}{\partial t^2} = c^2 \left( \frac{\partial^2 U}{\partial r^2} - 2 \left( \frac{\partial^2 U}{\partial r \partial s} \right) + \frac{\partial^2 U}{\partial s^2}  \right)
\end{align*}
Substituting these second derivatives in the general wave equation yields: \\
\begin{align*}
& \left[ c^2 \left( \frac{\partial^2 U}{\partial r^2} - 2 \left( \frac{\partial^2 U}{\partial r \partial s} \right) + \frac{\partial^2 U}{\partial s^2}  \right) \right] \\
-& c^2 \left[  \frac{\partial^2 U}{\partial r^2} + 2 \left( \frac{\partial^2 U}{\partial r \partial s} \right) + \frac{\partial^2 U}{\partial s^2}  \right] \\ \\
&=-4c^2 \left( \frac{\partial^2 U}{\partial r \partial s} \right) = 0 \\ \\
&\iff  \frac{\partial^2 U}{\partial r \partial s} = 0
\end{align*}
Our change-of-variables has transformed the original wave equation into this more workable form, which we can now simply integrate to extract $U$:
\begin{align*}
& \frac{\partial}{\partial r} \frac{\partial U}{\partial s}  = 0\\ \\
&\iff \frac{\partial U}{\partial s}  = C_1(s)\\ \\
&\iff U = C_2(s) + C_3(r)\\ \\
&\iff U(r,s) = F(r) + G(s)\\ \\
&\iff u(t,x) = F(x - ct) + G(x + ct)\\ \\
\end{align*}
We now apply our initial conditions.
\begin{align*}
u(0,x) = f(x) &\implies F(x) + G(x) = f(x) \\ \\
\left. \frac{\partial u}{\partial t} \right |_{(0,x)} = 0 &\implies -cF'(x) + cG'(x) = 0 \\
&\implies F'(x) = G'(x) \\
&\implies F(x) = G(x) + \delta \;\;\; \text{for some $\delta \in \mathbb{R}$} \\
\end{align*}
Our two constraints combine to imply
\begin{align*}
& 2G(x) + \delta = f(x) \\
&\implies G(x) = \frac{1}{2}(f(x) - \delta) \\
&\implies F(x) = \frac{1}{2}(f(x) + \delta)
\end{align*}
Hence
\begin{align*}
u(t,x) &= \frac{1}{2} (f(x-ct) + \delta) + \frac{1}{2}(f(x + ct) - \delta) \\ \\
&= \boxed{\frac{1}{2} f(x-ct)  + \frac{1}{2} f(x + ct)} \\ 
\end{align*}
\[
\boxed{
\begin{aligned}
u\left( \frac{1}{c}, x \right) &= \frac{1}{2} f(x-1) + \frac{1}{2} f(x + 1) \\ \\
u\left( \frac{2}{c}, x \right) &= \frac{1}{2} f(x-2) + \frac{1}{2} f(x + 2) \\ \\
u\left( \frac{3}{c}, x \right) &= \frac{1}{2} f(x-3) + \frac{1}{2} f(x + 3) \\ \\
\end{aligned}}
\]

\item Find the solution of \eqref{WE} for $x \in \mathbb{R}, t > 0$ with 
$$u(0,x) = 0 \quad \mathrm{and} \quad \frac{\partial u}{\partial t}(0,x) = g(x)$$
for $x \in \mathbb{R}$. In particular, write the solution in terms of the $g(x)$ function. Then, graph by hand $u(\frac{1}{c},x)$, $u(\frac{2}{c},x)$, and $u(\frac{3}{c},x)$ using different axes. \\ \\
\textcolor{blue}{\textbf{Solution} — Now we apply different initial conditions to the same general solution we obtained $u(t,x) = F(x-ct) + G(x+ct)$: \\
\begin{align*}
u(0,x) = 0 \implies& F(x) = -G(x) \\ \\
\left. \frac{\partial u}{\partial t} \right |_{(0,x)} = g(x) \implies& -cF'(x) + cG'(x) = g(x)
\end{align*}
From our first constraint, $F'(x) = -G'(x)$, so
\begin{align*}
-c(-G'(x)) + cG'(x) &= g(x) \\ \\
G'(x) &= \frac{g(x)}{2c} \\ \\
G(x) &= \frac{1}{2c} \int g(x)dx + \delta \\ \\
F(x) &= -\frac{1}{2c} \int g(x)dx - \delta
\end{align*} \\
Let $A(x)$ be the antiderivative of $g$, so that $A'(x) = g(x)$: \\
\begin{align*}
G(x) = \frac{1}{2c}A(x) + \delta &\;\;\;\;\;\;\;\;\; F(x) = -\frac{1}{2c} A(x) - \delta \\
\end{align*}
Hence
\begin{align*}
u(t,x) &=  \left( -\frac{1}{2c} A(x-ct) - \delta \right) + \left( \frac{1}{2c}A(x+ct) + \delta \right)  \\ \\
&=  \frac{1}{2c}  \left( \vphantom{\frac{1}{2c}} A(x+ct) - A(x-ct) \right)   \\ \\
&=  \boxed{\frac{1}{2c}  \int_{x-ct}^{x+ct} g(\zeta) d\zeta } \\ \\
\end{align*}
We compute a few solution functions at various times. First, note that the area under $g$ from $a$ to $b$ is
$$A(x) =
\begin{cases}
	0, & \mathrm{if} \ x < -1 \\ 
	x + 1, & \mathrm{if} \ |x| < 1 \\ 
	2, & \mathrm{if} \ x > 1
\end{cases}$$
Now, at different time increments $\frac{n}{c}$ for $n=1,2,3\cdots$, the solution is
\[
\begin{aligned}
u\left( \frac{n}{c}, x \right) &= \frac{1}{2c}  \int_{x-n}^{x+n} g(\zeta) d\zeta \\ \\
&= \frac{1}{2c} \left(\vphantom{\frac12}A(x+n) - A(x-n)\right) \\ \\
&= \frac{1}{2c} \left( \begin{cases}
	0, & \mathrm{if} \ x+n < -1 \\ 
	(x+n) + 1, & \mathrm{if} \ |x+n| < 1 \\ 
	2, & \mathrm{if} \ x+n > 1
\end{cases} - 
\begin{cases}
	0, & \mathrm{if} \ x-n < -1 \\ 
	(x-n) + 1, & \mathrm{if} \ |x-n| < 1 \\ 
	2, & \mathrm{if} \ x-n > 1
\end{cases}
 \right) \\ \\
&= \frac{1}{2c} \begin{cases}
	0 & \mathrm{if} \ x \leq -n-1 \\ 
	n+1+x, & \mathrm{if} \ -n -1 \leq x \leq -n + 1 \\
	2, & \mathrm{if} \ -n+1 \leq x < n-1 \\
	n+1 -x, & \mathrm{if} \ n - 1 \leq x \leq n + 1 \\
	0, & \mathrm{if} \ x \geq n + 1 \\ 
\end{cases} \\ \\
\end{aligned}
\]
Consider the symmetry of the $n+1+x$ and $n+1-x$ cases. If we apply the substitution $A = n-1$ and $B=n+1$, then these cases reduce to $$
\begin{cases}
B + x & \mathrm{if} \ -B \leq x \leq -A  \\
B - x & \mathrm{if} \ A \leq x \leq B
\end{cases}$$
Which reduces to $$
\begin{cases}
B - |x| & \mathrm{if} \ A \leq |x| \leq B  \\
\end{cases}$$
(Note that $B+x = B-|x|$ when $x$ is negative (in the first interval) and $B-x = B-|x|$ when $x$ is positive (in the second interval)). Also note that the two zero cases can be described by $|x| \geq n + 1$.
Hence
\[
\begin{aligned}
u\left( \frac{n}{c}, x \right) = \frac{1}{2c}
\begin{cases}
0 & \mathrm{if} \ |x| \geq n+1 \\
n+1 - |x| & \mathrm{if} \ n-1 \leq |x| \leq n+1 \\
2 & \mathrm{if} \ |x| \leq n - 1 \\
\end{cases}
\end{aligned}
\]
Hence
\[
\boxed{
\begin{aligned}
u\left( \frac{1}{c}, x \right) = \frac{1}{2c}
\begin{cases}
0 & \mathrm{if} \ |x| \geq 2 \\
2 - |x| & \mathrm{if} \ 0 \leq |x| \leq 2 \\
\end{cases}
\end{aligned}
}\]
\[
\boxed{
\begin{aligned}
u\left( \frac{2}{c}, x \right) = \frac{1}{2c}
\begin{cases}
0 & \mathrm{if} \ |x| \geq 3 \\
3 - |x| & \mathrm{if} \ 1 \leq |x| \leq 3 \\
2 & \mathrm{if} \ |x| \leq 1 \\
\end{cases}
\end{aligned}
}\]
\[
\boxed{
\begin{aligned}
u\left( \frac{3}{c}, x \right) = \frac{1}{2c}
\begin{cases}
0 & \mathrm{if} \ |x| \geq 4 \\
4 - |x| & \mathrm{if} \ 2 \leq |x| \leq 4 \\
2 & \mathrm{if} \ |x| \leq 2 \\
\end{cases}
\end{aligned}
}\]

\noindent $\textbf{2}$. (12 points) Let $f, g \in \mathrm{PC}^1[0,L]$ be given continuous functions. Find the solution of \eqref{WE}
for $0 < x < L$ and $t > 0$
with the boundary conditions
$$\frac{\partial u}{\partial x}(t,0) = \frac{\partial u}{\partial x}(t,L) =0 , \qquad t > 0$$
and the initial conditions for $0 < x < L$
$$ u(0,x) = f(x), \quad \frac{\partial u}{\partial t}(0,x) = g(x).$$ \\ \\
\textcolor{blue}{\textbf{Solution} — Since we now impose boundary conditions (in addition to initial conditions), we must now solve the wave equation on a bounded domain, which requires separation of variables. We assume the solution takes the separated form
$$u(t,x) = T(t)X(x)$$ For all $t,x \in \mathbb{R}$. Then $$\frac{\partial^2 u}{\partial t^2} = T''(t)X(x) \;\;\;\;\;\;\; \frac{\partial^2 u}{\partial x^2} = T(t)X''(x)$$ \\ and the wave equation takes the form \\
\begin{align*}
& T''(t)X(x) - c^2 T(t)X''(x) = 0 \implies \frac{T''(t)}{T(t)} = c^2 \frac{X''(x)}{X(x)} \\
\end{align*}
which is of the form $F(t) = G(x)$. Because we're assuming this equality holds for all $x,t$, then both sides of the equation must be constant (fix $x=x_0$ and we have $F(t) = G(x_0)$, which is equivalent to saying that $F(t)$ is (in general) equal to some constant $G(x_0)$. The same can be deduced by fixing $t=t_0$). Let $\lambda$ be this constant: $$\frac{T''(t)}{T(t)} = c^2 \frac{X''(x)}{X(x)} = \lambda \implies 
\begin{cases}
T'' - \lambda T = 0 \\
c^2X'' - \lambda X = 0 \\
\end{cases}$$
This is a system of independent second-order homogeneous linear ODEs, each of which can be solved with the characteristic equation to obtain the solutions \\ 
\begin{align*}
T(t) &= c_1 e^{-\sqrt{\lambda} t} + c_2 e^{\sqrt{\lambda}t} \\
X(x) &= c_2 e^{-\frac{\sqrt{\lambda}}{c} x} + c_3 e^{\frac{\sqrt{\lambda}}{c}x}
\end{align*}
When $\lambda > 0$, or
\begin{align*}
T(t) &= A + Bt \\
X(x) &= C + Dx
\end{align*}
When $\lambda = 0$, or 
\begin{align*}
T(t) &= A_1 \cos(\sqrt{-\lambda} t) + B_1 \sin(\sqrt{-\lambda} t)\\
X(x) &= C_1 \cos\left( \frac{\sqrt{-\lambda}}{c} x \right) + D_1 \sin \left( \frac{\sqrt{-\lambda}}{c} x \right)\\
\end{align*}
When $\lambda < 0$.
We will now apply our boundary conditions \\ $$\frac{\partial u}{\partial x}(t,0) = \frac{\partial u}{\partial x}(t,L) = 0$$ \\ Note that these are Neumann boundary conditions because they fix the physical slope of the string in space at the boundaries (as opposed to fixing the position of the string in space with Dirichlet boundaries). \\
$$\frac{\partial u}{\partial x}(t,0) = T(t)X'(x) \left. \right|_{(t,0)} = T(t)X'(0) = 0$$
$$\frac{\partial u}{\partial x}(t,L) = T(t)X'(x) \left. \right|_{(t,L)} = T(t)X'(L) = 0$$ \\
\textbf{For $\mathbf{\lambda > 0}$}: Ignoring the trivial case where $T(t) = 0$ (which causes the wave solution $u(t,x) = T(t)X(x)$ to be zero everywhere), these initial conditions reduce to \\
\begin{align*}
X'(0) = 0 \;\;\;\;\;\;\;\;\;& X'(L) = 0 \\
\implies c_2 = c_3 \;\;\;\;\;\;\;\;\;& \implies -c_2 e^{\frac{-\sqrt{\lambda}}{c} L} + c_3 e^{\frac{\sqrt{\lambda}}{c} L} = 0 \\
& \implies c_3 \left[2 \sinh{\frac{\sqrt{\lambda}}{c}L}\right] = 0 \\
& \implies c_3 = 0 \;\;\;\; \text{(because $\lambda \neq 0$, $c \neq 0$, $L \neq 0$)}\\
& \implies c_2 = 0 \\
\end{align*}
$$\implies X(x) = 0 \implies u(t,x) = 0$$
\textbf{For $\mathbf{\lambda = 0}$}:
\begin{align*}
X'(0) = 0 \;\;\;\;\;\;\;\;\;& X'(L) = 0 \\
\implies D = 0 \;\;\;\;\;\;\;\;\;& \implies D = 0 \\
& \implies X(x) = C \\
& \implies u(t,x) = C(A + Bt) \\
\end{align*}
\textbf{For $\mathbf{\lambda < 0}$}:
\begin{align*}
X'(0) = 0 \;\;\;\;\;\;\;\;\;& X'(L) = 0 \\
\implies \frac{D_1 \sqrt{-\lambda}}{c} = 0\;\;\;\;\;\;\;\;\;& \implies -C_1 \sin\left( \frac{\sqrt{-\lambda}}{c} L \right) + D_1 \cos \left( \frac{\sqrt{-\lambda}}{c} L \right) = 0\\
\implies D_1 = 0 \;\;\; \text{(since $\lambda \neq 0$, $c \neq 0$)}\;\;\;\;\;\;\;\;\;& \implies C_1 \sin\left( \frac{\sqrt{-\lambda}}{c} L \right) = 0 \;\;\; \text{(substituting $D_1 = 0$)}  \\
\;\;\;\;\;\;\;\;\;& \implies \frac{\sqrt{-\lambda}}{c} L = n\pi \;\;\; \text{(For nontrivial $C_1 \neq 0$)}  \\
\;\;\;\;\;\;\;\;\;& \implies \lambda_n = - \left( \frac{n\pi c}{L} \right) ^2 \\
\end{align*}
Hence $$T(t) = A_1 \cos \left( \frac{n\pi c}{L} t \right) + B_1 \sin \left( \frac{n \pi c}{L} t \right)$$
$$X(x) = C_1 \cos \left( \frac{n\pi x}{L} \right)$$ \\ and 
$$u(t,x) = \left[ A_1 \cos \left( \frac{n\pi c}{L} t \right) + B_1 \sin \left( \frac{n \pi c}{L} t \right) \right] C_1 \cos \left( \frac{n\pi x}{L} \right)$$ \\ 
By linearity of the wave equation, the superposition principle states that the general solution is the sum of all separated solution modes. Hence, the provided Neumann boundary conditions lead to the following general solution (over all possible $\lambda$)\\
$$\boxed{u(t,x) = a_0 + b_0 t + \sum_{n=1}^{\infty} \left[ a_n \cos \left( \frac{n\pi c}{L} t \right) + b_n \sin \left( \frac{n \pi c}{L} t \right) \right]  \cos \left( \frac{n\pi x}{L} \right)} $$ \\
We now apply the initial conditions:
$$u(0,x) = f(x) \implies  a_0  + \sum_{n=1}^{\infty} a_n \cos \left( \frac{n\pi x}{L} \right) = f(x)$$
$$\left. \frac{\partial u}{\partial t} \right|_{(0,x)}  = g(x) \implies b_0  + \left( \frac{\pi c}{L} \right) \sum_{n=1}^{\infty} n b_n  \cos \left( \frac{n\pi x}{L} \right) = g(x)$$ \\
Hence the $a_n$ coefficients must be exactly the cosine-series coefficients of the Fourier expansion of $f(x)$ on $[0,L]$. Likewise, the the velocity condition, the product $(n\pi c / L) b_n$ must be exactly the cosine-series coefficients of the Fourier expansion of $g(x)$ on $[0,L]$. Note that these two series must exist due to $f,g \in PC^1[0,L]$\\ \\
$$\boxed{a_0 = \frac{1}{L} \int_{0}^{L} f(x) dx}$$
$$\boxed{a_n = \frac{2}{L} \int_{0}^{L} f(x) \cos \left( \frac{n\pi x}{L} \right) dx}$$
$$\boxed{b_0 = \frac{1}{L} \int_{0}^{L} g(x) dx}$$
$$\frac{n\pi c}{L} b_n = \frac{2}{L} \int_{0}^{L} g(x) \cos \left( \frac{n\pi x}{L} \right) dx \implies \boxed{b_n = \frac{2}{n\pi c} \int_{0}^{L} g(x) \cos \left( \frac{n\pi x}{L} \right) dx}$$ \\
}
