\noindent $\textbf{1}$. (14 points) Let $a,b > 0$ and $g \in \mathrm{PC}^1[0,a]$ be given. Find the solution $u(x,y)$ of 
$$\Delta u = 0$$
on the rectangle $ \Omega = [0,a] \times [0,b],$
with
$$\frac{\partial u}{\partial x}(0,y) = \frac{\partial u}{\partial x} (a,y) = 0$$
for $0 < y < b$ and
$$\frac{\partial u}{\partial y}(x,0) = 0 \qquad \mathrm{and} \qquad u(x,b) = g(x)$$
for $0 < x < a$.

\noindent \underline{\emph{Hint:}} It might be easiest to use $\sinh$ and $\cosh$ in some places rather than real exponentials. \\ \\
\textcolor{blue}{\textbf{Solution} — 
\begin{align*}
\Delta u = 0 \implies& \frac{\partial^2 u}{\partial x^2} + \frac{\partial^2 u}{\partial y^2} = 0
\end{align*}
Since this is a linear PDE, we proceed with separation of variables, assuming the solution takes the (separated) form $u(x,y) = X(x)Y(y)$ \\
\begin{align*}
\implies& X''(x)Y(y) = -X(x)Y''(y) \\
\implies& \frac{X''(x)}{X(x)} = -\frac{Y''(y)}{Y(y)} = \lambda
\end{align*} \\
This equation takes the form $F(x) = G(y)$ for all $x,y$ $\implies$ both must be equal to some constant $\lambda$ (fix $x = x_0$, and then $F(x_0) = G(y)$ for all $y \implies G(y)$ is constant $\implies F(x)$ is constant since they're equal to each other). This yields the two ordinary differential equations
\begin{align*}
\begin{cases}
X'' - \lambda X = 0 \\
Y'' + \lambda Y = 0
\end{cases}
\end{align*} \\
whose solutions depend on the sign of $\lambda$. \\ \\
\textbf{For $\lambda < 0$}: \\ \\
$$X(x) = c_0 \cos \left( \sqrt{-\lambda} x \right) + c_1 \sin \left( \sqrt{-\lambda} x \right) \;\;\;\;\;\;\;\;\; Y(y) = c_2 e^{\sqrt{-\lambda} y} + c_3 e^{-\sqrt{-\lambda} y}$$
\textbf{For $\lambda = 0$}: \\ \\
$$X(x) = Ax + B \;\;\;\;\;\;\;\;\; Y(y) = Cy + D$$
\textbf{For $\lambda > 0$}: \\ \\
$$X(x) = c_4 e^{\sqrt{\lambda} x} + c_5 e^{-\sqrt{\lambda} x} \;\;\;\;\;\;\;\;\; Y(y) = c_6 \cos \left( \sqrt{\lambda} y \right) + c_7 \sin \left( \sqrt{\lambda} y \right)$$ \\
We apply the boundary conditions to $u(x,y) = X(x)Y(y)$: \\ \\
\begin{align*}
\frac{\partial u}{\partial x}(0,y) = 0 \implies& X'(0)Y(y) = 0 \\
\implies& X'(0) = 0 \;\;\;\; \text{(For $Y(y) \neq 0$)}
\end{align*}
\begin{align*}
\frac{\partial u}{\partial x}(a,y) = 0 \implies& X'(a)Y(y) = 0 \\
\implies& X'(a) = 0 \;\;\;\; \text{(For $Y(y) \neq 0$)}
\end{align*}
We use these results to constrain the unknown constants in our separated solutions: \\ \\
\textbf{For $\lambda < 0$}: \\ \\
\begin{align*}
X'(0) = 0 \implies& \sqrt{-\lambda} c_1 = 0 \\
\implies& \boxed{c_1 = 0} \\
X'(a) = 0 \implies& -\sqrt{-\lambda} c_0 \sin \left( \sqrt{-\lambda} a \right) = 0 \\
\implies& \sin \left( \sqrt{-\lambda} a \right) = 0 \;\;\;\; \text{(for nontrivial solution with $c_0 \neq 0$)} \\
\implies& \sqrt{-\lambda} a = n\pi \;\;\;\;\; \text{for $n \in \mathbb{Z}$} \\
\implies& \boxed{\lambda = - \left( \frac{n \pi}{a} \right) ^2}
\end{align*}
\textbf{For $\lambda = 0$}: \\
\begin{align*}
X'(0) = 0, X'(a) = 0 \implies& \boxed{A = 0}\\
\end{align*}
\textbf{For $\lambda > 0$}: \\ \\
\begin{align*}
X'(0) = 0 \implies& \sqrt{\lambda} c_4 - \sqrt{\lambda} c_5 = 0 \\
\implies& \boxed{c_4 = c_5} \\
X'(a) = 0 \implies& \sqrt{\lambda} c_4 e^{\sqrt{\lambda} a} - \sqrt{\lambda} c_5 e^{-\sqrt{\lambda} a} = 0 \\
\implies& \sqrt{\lambda} c_4 \left( \underbrace{ e^{\sqrt{\lambda} a} - e^{-\sqrt{\lambda} a} }_{\neq 0} \right) = 0 \;\;\; \text{(using $c_4 = c_5$)} \\
\implies& \boxed{c_4 = 0}
\end{align*}
This leads us to general separated solutions: \\ \\
\textbf{For $\lambda < 0$}: \\ \\
$$u(x,y) = \cos \left( \frac{n \pi x}{a} \right) \left( A_n e^{ \frac{n\pi y}{a} } + B_n e^{-\frac{n\pi y}{a}}  \right)$$
\textbf{For $\lambda = 0$}: \\ \\
$$u(x,y) = \alpha y + \beta$$
\textbf{For $\lambda > 0$}: \\ \\
$$u(x,y) = 0$$
Hence the general vertically-constrained solution is the superposition of all these separated solutions: \\
$$u(x,y) = \alpha y + \beta + \sum_{n=1}^{\infty} \cos \left( \frac{n \pi x}{a} \right) \left( A_n e^{ \frac{n\pi y}{a} } + B_n e^{-\frac{n\pi y}{a}}  \right)$$ \\
We proceed to apply the horizontal boundary conditions. \\
\begin{align*}
\frac{\partial u}{\partial y}(x,0) = 0 \implies&  \left. \alpha + \sum_{n=1}^{\infty} \frac{n\pi}{a} \cos \left( \frac{n \pi x}{a} \right) \left( A_n e^{ \frac{n\pi y}{a} } - B_n e^{-\frac{n\pi y}{a}} \right) \right|_{(x,0)} = 0 \\
\implies& \alpha + \sum_{n=1}^{\infty} \frac{n\pi}{a} (A_n - B_n) \cos \left( \frac{n \pi x}{a} \right) = 0  \\
\end{align*}
Hence $\alpha$ is the constant term and each $n \pi (A_n - B_n)  / a$ is the Fourier coefficient of the cosine expansion of $f(x) = 0$ on $x\in[0,a]$. Since the average value of this function is $0$ everyone, we have that $\boxed{\alpha = 0}$. Also, \\
\begin{align*}
\implies& \frac{n\pi (A_n - B_n) }{a} = \frac{2}{a} \int_0^a (-\alpha) \cos \left( \frac{n\pi x}{a} \right) dx \\
\implies& (A_n - B_n)  = -\frac{2\alpha}{n\pi} \int_0^a \cos \left( \frac{n\pi x}{a} \right) dx = 0 \\
\implies & \boxed{A_n = B_n}
\end{align*} \\
Hence we can write the solution as
$$u(x,y) =  \beta + 2 \sum_{n=1}^{\infty} C_n \cos \left( \frac{n \pi x}{a} \right) \cosh \left( \frac{n \pi y}{a} \right)$$ \\
We apply the final boundary condition
\begin{align*}
u(x,b) = g(x) \implies& \beta + 2 \sum_{n=1}^{\infty} C_n \cos \left( \frac{n \pi x}{a} \right) \cosh \left( \frac{n \pi b}{a} \right) = g(x) \\
\implies& g(x) = \beta + \sum_{n=1}^{\infty} \left[ \underbrace{ 2C_n \cosh \left( \frac{n \pi b}{a} \right) }_{\zeta_n} \right] \cos \left( \frac{n \pi x}{a} \right) \\
\implies& g(x) = \beta + \sum_{n=1}^{\infty} \zeta_n \cos \left( \frac{n \pi x}{a} \right)
\end{align*}
Which is the cosine-expansion of $g(x)$ on $[0,a]$, with $\beta$ being the average value of $g(x)$ on $[0,a]$. Hence
\begin{align*}
\boxed{\beta = \frac{1}{a} \int_0^a g(x) dx} \\ \\
\end{align*}
And
\begin{align*}
\zeta_n &= \frac{2}{a} \int_0^a g(x) \cos \left( \frac{n \pi x}{a} \right) dx \\
\implies& 2C_n \cosh \left( \frac{n \pi b}{a} \right) = \frac{2}{a} \int_0^a g(x) \cos \left( \frac{n \pi x}{a} \right) dx \\
\implies& \boxed{C_n = \frac{1}{a} \operatorname{sech} \left( \frac{n \pi b}{a} \right) \int_0^a g(x) \cos \left( \frac{n \pi x}{a} \right) dx }
\end{align*}
In the general solution
$$\boxed{u(x,y) =  \beta + 2 \sum_{n=1}^{\infty} C_n \cos \left( \frac{n \pi x}{a} \right) \cosh \left( \frac{n \pi y}{a} \right)}$$
}

\noindent $\textbf{2}$. (16 points) Let $a > 0$ be given and find (using polar coordinates) the solution $u(r,\theta)$ of Laplace's equation
$$\Delta u = 0$$
on the wedge
$$ \Omega = \left \{ (r,\theta) : 0 \leq r < a, 0 \leq \theta \leq \frac{\pi}{3} \right \}$$
with
$$u(r,0) = u\left (r,\frac{\pi}{3} \right)  = 0$$
for $0 \leq r < a$ and
$$u(a,\theta) = a^2$$
for $0 \leq \theta \leq \frac{\pi}{3}$.
In particular, make sure to only find solutions that remain bounded as $r \to 0$. \\ \\
\textcolor{blue}{\textbf{Solution} — Using polar coordinates conveniently transforms the (non-uniform) boundary of the wedge to a uniform boundary (a rectangle) in polar space. Hence, we must also express the laplace equation in polar form. We apply the change-of-variables $$x(r, \theta) = r \cos \theta \;\;\;\;\;\;\; y(r, \theta) = r \sin \theta$$ \\
And express $$u(x,y) \implies u(r,\theta)$$ \\
keeping in mind that $r$ and $\theta$ are both functions of $x$ and $y$. We proceed to find $\frac{\partial^2 u}{\partial x^2}$ and $\frac{\partial^2 u}{\partial y^2}$ in terms of $\theta$ and $r$. $$\frac{\partial u}{\partial x} = \frac{\partial u}{\partial r} \frac{\partial r}{\partial x} + \frac{\partial u}{\partial \theta} \frac{\partial \theta}{\partial x}$$ \\
To find $\frac{\partial r}{\partial x}$ we need $r$ as a function of $x$ and $y$ : $r(x,y)$. We use
\begin{align*}
r^2 = x^2 + y^2 \implies& 2r \frac{\partial r}{\partial x} = 2x \\
\implies& \frac{\partial r}{\partial x} = \frac{x}{r} \\
\implies& \boxed{\frac{\partial r}{\partial x} = \cos \theta}
\end{align*}
To find $\frac{\partial \theta}{\partial x}$ we need $\theta(x,y)$. We use
\begin{align*}
\tan \theta = \frac{y}{x} \implies& \sec^2 \theta \frac{\partial \theta}{\partial x} = -\frac{y}{x^2} \\
\implies& \frac{\partial \theta}{\partial x} = - \cos^2 \theta \frac{y}{x^2} \\
\implies& \frac{\partial \theta}{\partial x} = - \cos^2 \theta \frac{r \sin \theta}{(r \cos \theta)^2} \\
\implies& \boxed{\frac{\partial \theta}{\partial x} = - \frac{\sin \theta}{r}} \\
\end{align*}
Hence $$\frac{\partial u}{\partial x} = \cos \theta \frac{\partial u}{\partial r} - 
\frac{\sin \theta}{r} \frac{\partial u}{\partial \theta}$$
And 
\begin{align*}
\frac{\partial^2 u}{\partial x^2} &= \frac{\partial}{\partial x} \left( \cos \theta \frac{\partial u}{\partial r} - \frac{\sin \theta}{r} \frac{\partial u}{\partial \theta} \right) \\
&= \underbrace{ \frac{\partial (\cos \theta)}{\partial x} }_{\text{A}} \frac{\partial u}{\partial r} + \cos \theta \underbrace{ \frac{\partial}{\partial x} \frac{\partial u}{ \partial r} }_{\text{B}} - \underbrace{ \frac{\partial (\frac{\sin \theta}{r})}{\partial x} }_{\text{C}} \frac{\partial u}{\partial \theta} - \frac{\sin \theta}{r} \underbrace{ \frac{\partial }{\partial x}\frac{\partial u}{\partial \theta} }_{\text{D}}
\end{align*} \\
Simplifying these four pieces: \\ \\
\textbf{A}: \\
\begin{align*}
\frac{\partial (\cos \theta)}{\partial x} &= \frac{\partial}{\partial x} \left( \frac{x}{\sqrt{x^2 + y^2}} \right) \\ \\
&= \frac{\sqrt{x^2 + y^2} - \frac{x^2}{\sqrt{x^2 + y^2}}}{x^2 + y^2} \\ \\
&= \frac{r - r\cos^2 \theta}{r^2} \\ \\
&= \frac{\sin^2 \theta}{r} \\ \\
\end{align*}
\textbf{B}: Note that $\partial u / \partial r$ is some function of $r,\theta$, so let's temporarily define it as the dummy function $\phi(r,\theta)$. Hence, \\
\begin{align*}
\frac{\partial }{\partial x} \frac{\partial u}{\partial r}  &= \frac{\partial }{\partial x} \phi(r, \theta) \\ \\
 &=  \frac{\partial \phi}{\partial r} \frac{\partial r}{\partial x} + \frac{\partial \phi}{\partial \theta} \frac{\partial \theta}{\partial x} \\ \\
 &= \frac{\partial^2 u}{\partial r^2} \frac{\partial r}{\partial x} + \frac{\partial^2 u}{\partial \theta \partial r} \frac{\partial \theta}{\partial x} \\ \\
 &= \cos \theta \frac{\partial^2 u}{\partial r^2} - \frac{\sin \theta}{r}  \frac{\partial^2 u}{\partial \theta \partial r} 
\end{align*} \\
\textbf{C}: \\ \\
\begin{align*}
 \frac{\partial (\frac{\sin \theta}{r})}{\partial x} &= \frac{\partial}{\partial x} \left( \frac{y}{x^2 + y^2} \right) \\ \\
 &= \frac{-2xy}{(x^2 + y^2)^2} \\ \\ 
 &= \frac{-2r^2 \sin \theta \cos \theta}{r^4} \\ \\
 &= \frac{-2 \sin \theta \cos \theta}{r^2} \\ \\
 &= -\frac{\sin 2 \theta}{r^2} \\ \\
\end{align*}
\textbf{D}: Note that $\partial u / \partial \theta$ is some function $r, \theta$, so let's also temporarily define it as the dummy function $\phi(r, \theta)$. Then
\begin{align*}
\frac{\partial }{\partial x}\frac{\partial u}{\partial \theta} &= \frac{\partial }{\partial x} \phi(r, \theta) \\ \\
&= \frac{\partial \phi}{\partial r} \frac{\partial r}{\partial x} + \frac{\partial \phi}{\partial \theta} \frac{\partial \theta}{\partial x} \\ \\
&= \frac{\partial^2 u}{\partial r \partial \theta} \frac{\partial r}{\partial x} + \frac{\partial^2 u}{\partial \theta^2} \frac{\partial \theta}{\partial x} \\ \\
&= \cos \theta \frac{\partial^2 u}{\partial r \partial \theta} - \frac{\sin \theta}{r}  \frac{\partial^2 u}{\partial \theta^2}
\end{align*}
Hence \\
\begin{align*}
\frac{\partial^2 u}{\partial x^2} =& \frac{\sin^2 \theta}{r} \frac{\partial u}{\partial r} + \cos \theta \left[ \cos \theta \frac{\partial^2 u}{\partial r^2} - \frac{\sin \theta}{r} \frac{\partial^2 u}{\partial \theta \partial r} \right] \\
+&  \frac{\sin 2\theta}{r^2} \frac{\partial u}{\partial \theta} - \frac{\sin \theta}{r} \left[ \cos \theta \frac{\partial^2 u}{\partial r \partial \theta} - \frac{\sin \theta}{r} \frac{\partial^2 u}{\partial \theta^2}\right]
\end{align*}
\begin{align*}
\frac{\partial^2 u}{\partial x^2} = \cos^2 \theta \frac{\partial^2 u}{\partial r^2} + \frac{\sin^2 \theta}{r} \frac{\partial u}{\partial r} - \frac{\sin 2 \theta}{r} \frac{\partial^2 u}{\partial r \partial \theta} + \frac{\sin^2 \theta}{r^2} \frac{\partial^2 u}{\partial \theta^2} + \frac{\sin 2\theta}{r^2} \frac{\partial u}{\partial \theta}
\end{align*}
An analogous process yields
\begin{align*}
\frac{\partial^2 u}{\partial y^2} = \sin^2 \theta \frac{\partial^2 u}{\partial r^2} + \frac{\cos^2 \theta}{r} \frac{\partial u}{\partial r} + \frac{\sin 2 \theta}{r} \frac{\partial^2 u}{\partial r \partial \theta} + \frac{\cos^2 \theta}{r^2} \frac{\partial^2 u}{\partial \theta^2} - \frac{\sin 2\theta}{r^2} \frac{\partial u}{\partial \theta}
\end{align*}
Hence $$\frac{\partial^2 u}{\partial x^2} + \frac{\partial^2 u}{\partial y^2} = \frac{\partial^2 u}{\partial r^2} + \frac{1}{r} \frac{\partial u}{\partial r} + \frac{1}{r^2} \frac{\partial u^2}{\partial \theta^2}$$ \\
So the polar Laplacian equation must be
$$\boxed{\frac{\partial^2 u}{\partial r^2} + \frac{1}{r} \frac{\partial u}{\partial r} + \frac{1}{r^2} \frac{\partial^2 u}{\partial \theta^2} = 0}$$
Now we proceed to find the general solution of this differential equation, using the same separation of variables method as before. We assume $u(r,\theta)$ takes the form $R(r)\Theta(\theta)$. Then 
$$R''(r) \Theta(\theta) +\frac{R'(r) \Theta(\theta)}{r} + \frac{R(r) \Theta''(\theta)}{r^2} = 0$$
Dividing through by $R(r) \Theta(\theta)$ yeilds \\
$$\frac{R''(r)}{R(r)} +\frac{R'(r)}{r R(r)} + \frac{\Theta''(\theta)}{r^2 \Theta(\theta)} = 0$$
$$\implies \underbrace{ r^2 \frac{R''(r)}{R(r)} + r\frac{R'(r)}{R(r)} }_{F(r)}= \underbrace {- \frac{\Theta''(\theta)}{\Theta(\theta)} }_{G(\theta)} = \mu$$ \\
The left and right are functions of $r$ only, and $\theta$ only, respectively. So our differential equal that we assume works for all $r, \theta$ implies this equation, which, also, supposedly works for all $r, \theta$. And since $F(r) = G(\theta)$ for all $r, \theta$, then both sides must be equal to the same constant (fix $\theta = \theta_0$ and we get $F(r) = G(\theta_0)$ for all $r$, the definition of a constant function). We call this constant (the separation constant) $\mu$.
$$ \implies \begin{cases}
r^2 R'' + r R' - \mu R = 0 \\
\Theta'' + \mu \Theta= 0
\end{cases}$$
Since the radial equation is non-trivial, let us solve the easier angular equation in its entirety first. By the characteristic equation, we obtain \\ \\
\textbf{For $\mu > 0$}: \\ \\
$$\Theta(\theta) = c_0 \cos \left( \sqrt{\mu} \theta \right) + c_1 \sin \left( \sqrt{\mu} \theta \right)$$
\textbf{For $\mu = 0$}: \\ \\
$$\Theta(\theta) = c_2 + c_3 \theta$$
\textbf{For $\mu < 0$}: \\ \\
$$\Theta(\theta) = c_4 e^{\sqrt{-\mu} \theta} +c_5 e^{-\sqrt{-\mu} \theta} $$  \\
We apply the angular Neumann boundary condition
\begin{align*}
& u(r,0) = u\left (r,\frac{\pi}{3} \right)  = 0 \\ 
\implies& R(r) \Theta(0) = 0 \implies \Theta(0) = 0 \;\;\; \text{(For nontrivial $R(r)$)} \\
\implies& R(r) \Theta \left( \frac{\pi}{3} \right) = 0 \implies \Theta \left( \frac{\pi}{3} \right) = 0 \;\;\; \text{(For nontrivial $R(r)$)}
\end{align*}
\textbf{For $\mu > 0$}: \\ \\
\begin{align*}
\Theta(0) = 0 \implies& \boxed{c_0 = 0} \\
\Theta \left( \frac{\pi}{3} \right) = 0 \implies& c_1 \sin \left( \sqrt{\mu} \frac{\pi}{3} \right) = 0 \\
\implies& \sin \left( \sqrt{\mu} \frac{\pi}{3} \right) = 0 \;\;\; \text{(for $c_1 \neq 0$)} \\
\implies& \frac{\pi \sqrt{\mu}}{3} = n\pi \;\;\; \text{For $n \in \mathbb{Z}$} \\
\implies& \boxed{\mu = (3n)^2}
\end{align*}
\textbf{For $\mu = 0$}: \\ \\
\begin{align*}
\Theta(0) = 0 \implies& \boxed{c_2 = 0} \\
\Theta \left( \frac{\pi}{3} \right) = 0 \implies& \boxed{c_3 = 0}
\end{align*}
\textbf{For $\mu < 0$}: \\ \\
\begin{align*}
\Theta(0) = 0 \implies& \boxed{c_4 = -c_5}\\
\Theta \left( \frac{\pi}{3} \right) = 0 \implies& c_4 e^{\sqrt{-\mu} \frac{\pi}{3}} +c_5 e^{-\sqrt{-\mu} \frac{\pi}{3}} = 0 \\
\implies& c_4 \left( e^{\sqrt{-\mu} \frac{\pi}{3}} - e^{-\sqrt{-\mu} \frac{\pi}{3}} \right) = 0 \;\;\; \text{(substituting $c_5 = -c_4$)} \\
\implies& 2 c_4 \underbrace{ \sinh \left( \sqrt{-\mu} \frac{\pi}{3} \right) }_{\neq 0} = 0 \\
\implies& \boxed{c_4 = 0}
\end{align*}
We obtain trivial solutions for everything except the $\mu > 0$ case, so that case is the only contributor of angular eigenfunction solutions:
$$\Theta_n(\theta) = c_1 \sin \left(3n\theta \right)$$
Let us now solve the radial equation, given that we know $\mu = (3n)^2$. We must hence solve
$$r^2 R'' + r R' - (3n)^2 R = 0$$
This is linear but has \textbf{non}-constant coefficients, making it impossible to solve using the characteristic equation. Instead, this has the form of a Cauchy-Euler differential equation, which suggests the solution guess $R(r) = r^k$. Plugging in yields
\begin{align*}
& k(k-1) r^k + k r^k - (3n)^2 r^k = 0 \\
\implies& k(k-1) + k - (3n)^2 = 0 \\
\implies& k^2 - (3n)^2 = 0 \\
\implies& k = 3n \\
\end{align*}
Hence the general solution is
